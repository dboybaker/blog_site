{{/* File embed shortcode for displaying file contents dynamically in a collapsible block. Usage: {{< file-embed path="relative/path/to/file.py" lang="python" label="filename.py" >}} {{< file-embed path="relative/path/to/file.txt" lang="text" label="dump.txt" summary="The raw dump" >}} Parameters: - path: Path to the file relative to the site root (required) - lang: Language for syntax highlighting (default: "text") - label: Display label for the toggle (default: filename from path) - summary: Lead text on the toggle (default: "Full training script", kept for the article that shipped with it - pass a summary for any other use) */}} {{ $path := .Get "path" }} {{ $lang := .Get "lang" | default "text" }} {{ $label := .Get "label" | default (path.Base $path) }} {{ $summary := .Get "summary" | default "Full training script" }} {{ if not $path }} {{ errorf "file-embed shortcode: 'path' parameter is required" }} {{ end }} {{ $fileContent := readFile $path }} {{ if not $fileContent }} {{ errorf "file-embed shortcode: could not read file '%s'" $path }} {{ end }}
{{ $summary }} — click to expand {{ $label }} {{ highlight $fileContent $lang "" }}
